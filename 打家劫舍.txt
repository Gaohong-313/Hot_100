class Solution {
public:
    int rob(vector<int>& nums) {
        int n = nums.size();
        // 处理极端边界情况
        if (n == 0) return 0;
        if (n == 1) return nums[0];

        // 用两个变量代替数组
        // prev2 相当于 result[i-2]
        int prev2 = nums[0]; 
        // prev1 相当于 result[i-1]
        int prev1 = max(nums[0], nums[1]); 

        for (int i = 2; i < n; i++) {
            // 计算当前的最大值
            int curr = max(prev1, prev2 + nums[i]);
            // 滚动更新变量
            prev2 = prev1;
            prev1 = curr;
        }
        return prev1;
    }
};
