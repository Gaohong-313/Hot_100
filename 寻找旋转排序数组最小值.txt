class Solution {
public:
    int findMin(vector<int>& nums) {
        int left = 0;
        int right = nums.size() - 1;

        // 当 left < right 时，说明搜索区间内不止一个元素
        while (left < right) {
            int mid = left + (right - left) / 2;

            // 关键判断：与右边界比较
            if (nums[mid] > nums[right]) {
                // nums[mid] 在左侧升序数组，最小值在 mid 右侧
                left = mid + 1;
            } else {
                // nums[mid] 在右侧升序数组，最小值是 mid 或在 mid 左侧
                right = mid;
            }
        }

        // 循环结束时，left == right，指向最小值
        return nums[left];
    }
};
