class Solution {
private:
    vector<vector<string>> result;
    vector<string> path; // 存放当前分割出的回文串

    // 判断字符串 s 的 [start, end] 区间是否是回文串
    bool isPalindrome(const string& s, int start, int end) {
        for (int i = start, j = end; i < j; i++, j--) {
            if (s[i] != s[j]) {
                return false;
            }
        }
        return true;
    }

    // 回溯函数
    // startIndex: 当前切割的起始位置
    void backtracking(const string& s, int startIndex) {
        // 终止条件：如果起始位置到了字符串末尾，说明找到了一组解
        if (startIndex >= s.size()) {
            result.push_back(path);
            return;
        }

        // 单层搜索逻辑
        for (int i = startIndex; i < s.size(); i++) {
            // 1. 判断当前截取的子串 s[startIndex...i] 是否是回文
            if (isPalindrome(s, startIndex, i)) {
                // 2. 是回文，则截取并加入路径
                string str = s.substr(startIndex, i - startIndex + 1);
                path.push_back(str);
            } else {
                // 3. 不是回文，直接跳过，不进行后续递归
                continue;
            }

            // 4. 递归：处理剩下的字符串，起始位置变为 i + 1
            backtracking(s, i + 1);

            // 5. 回溯：撤销刚才的选择
            path.pop_back();
        }
    }

public:
    vector<vector<string>> partition(string s) {
        result.clear();
        path.clear();
        backtracking(s, 0);
        return result;
    }
};
