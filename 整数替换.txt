class Solution {
public:
    int integerReplacement(long long n) {
        int count = 0;
        while (n > 1) {
            if (n % 2 == 0) {
                // 偶数直接除2
                n /= 2;
            } else {
                // 奇数情况
                if (n == 3) {
                    // 特判：3 必须减1，否则 3->4->2->1 会多一步
                    n--;
                } else if (n % 4 == 1) {
                    // 例如 5, 9, 13... 减1后能被4整除，选 n-1
                    n--;
                } else {
                    // n % 4 == 3，例如 7, 11, 15... 加1后能被4整除，选 n+1
                    n++;
                }
            }
            count++;
        }
        return count;
    }
};
