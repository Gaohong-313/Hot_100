class Solution {
public:
    int maxProduct(vector<int>& nums) {
        if (nums.empty()) return 0;
        
        // 初始化：当前最大值、当前最小值、全局最大值都设为第一个元素
        int currMax = nums[0];
        int currMin = nums[0];
        int result = nums[0]; 
        
        for (int i = 1; i < nums.size(); ++i) {
            // 【关键坑点】因为 currMax 马上要被更新，必须先保存旧值
            // 否则计算 currMin 时会用到更新后的 currMax，导致逻辑错误
            int prevMax = currMax; 
            
            // 状态转移：取三者中的最大值（当前元素本身、之前的最大值*当前元素、之前的最小值*当前元素）
            currMax = max(nums[i], max(prevMax * nums[i], currMin * nums[i]));
            
            // 状态转移：取三者中的最小值
            currMin = min(nums[i], min(prevMax * nums[i], currMin * nums[i]));
            
            // 更新全局最大乘积
            result = max(result, currMax);
        }
        return result;
    }
};
