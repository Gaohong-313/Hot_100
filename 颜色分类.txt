class Solution {
public:
    void sortColors(vector<int>& nums) {
        vector<int> count(3, 0);
        
        // 1. 统计各个颜色的数量
        for(int i = 0; i < nums.size(); i++){
            count[nums[i]]++; // 这样写比三个 if 更简洁
        }
        
        // 2. 重新填充数组
        int index = 0;
        while(count[0] > 0){ nums[index++] = 0; count[0]--; }
        while(count[1] > 0){ nums[index++] = 1; count[1]--; }
        while(count[2] > 0){ nums[index++] = 2; count[2]--; } // 修复 Bug
    }
};
