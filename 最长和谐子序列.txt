class Solution {
public:
    int findLHS(vector<int>& nums) {
        unordered_map<int, int> countMap;
        
        // 1. 统计每个数字出现的频次
        for (int num : nums) {
            countMap[num]++;
        }
        
        int result = 0;
        // 2. 遍历哈希表，寻找相邻的两个数值
        for (auto& [num, count] : countMap) {
            // 如果 num + 1 也存在，说明可以组成和谐子序列
            if (countMap.find(num + 1) != countMap.end()) {
                result = max(result, count + countMap[num + 1]);
            }
        }
        return result;
    }
};
