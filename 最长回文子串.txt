class Solution {
public:
    int longestPalindromeSubseq(std::string s) {
        int n = s.length();
        // 创建 n x n 的二维 dp 数组，初始化为 0
        std::vector<std::vector<int>> dp(n, std::vector<int>(n, 0));

        // 初始化：单个字符的回文子序列长度为 1
        for (int i = 0; i < n; ++i) {
            dp[i][i] = 1;
        }

        // 状态转移
        // i 从后往前遍历，保证计算 dp[i][j] 时，dp[i+1][...] 已经计算过
        for (int i = n - 1; i >= 0; --i) {
            // j 从 i+1 开始往后遍历
            for (int j = i + 1; j < n; ++j) {
                if (s[i] == s[j]) {
                    // 两端字符相等，长度加 2
                    dp[i][j] = dp[i + 1][j - 1] + 2;
                } else {
                    // 两端字符不等，取去掉左端或去掉右端后的最大值
                    dp[i][j] = std::max(dp[i + 1][j], dp[i][j - 1]);
                }
            }
        }

        // 返回整个字符串区间 [0, n-1] 的最长回文子序列长度
        return dp[0][n - 1];
    }
};
