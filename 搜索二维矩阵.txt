class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        // 1. 处理空矩阵的边界情况
        if (matrix.empty() || matrix[0].empty()) {
            return false;
        }

        int m = matrix.size();
        int n = matrix[0].size();
        
        // 2. 确定目标所在的行
        int target_row = -1;
        
        // 遍历每一行，找到第一个元素小于等于 target 的行
        // 注意：我们要找的是“最后一个”满足 matrix[i][0] <= target 的行
        for (int i = 0; i < m; ++i) {
            if (matrix[i][0] <= target) {
                target_row = i;
            } else {
                // 如果当前行的首元素已经大于 target，后面的行肯定也更大，直接跳出
                break;
            }
        }

        // 如果没有找到合适的行（例如 target 比所有行的首元素都小）
        if (target_row == -1) {
            return false;
        }

        // 3. 在确定的行中进行二分查找
        return middlesearch(matrix[target_row], target);
    }

    bool middlesearch(vector<int> &nums, int target) {
        int left = 0;
        int right = nums.size() - 1;
        while (left <= right) {
            int middle = left + (right - left) / 2; // 防止溢出的写法
            if (nums[middle] < target) {
                left = middle + 1;
            } else if (nums[middle] > target) {
                right = middle - 1;
            } else {
                return true;
            }
        }
        return false;
    }
};
