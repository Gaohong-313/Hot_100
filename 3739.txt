typedef long long LL;

// 树状数组模板
class Fenwick {
private:
    vector<int> sums_;
public:
    explicit Fenwick(int n) : sums_(n + 1) {}
    
    int lowbit(int x) { return x & -x; }
    
    void add(int idx, int val) {
        for (; idx < sums_.size(); idx += lowbit(idx)) {
            sums_[idx] += val;
        }
    }
    
    int query(int idx) {
        int ans = 0;
        for (; idx > 0; idx -= lowbit(idx)) {
            ans += sums_[idx];
        }
        return ans;
    }
};

class Solution {
public:
    long long countMajoritySubarrays(vector<int>& nums, int target) {
        int n = nums.size();
        // 1. 离散化处理：因为 2 * presum[i] - i 可能为负数，树状数组下标必须 >= 1
        vector<int> vals{0}; 
        int presum = 0;
        for (int i = 1; i <= n; i++) {
            presum += (nums[i - 1] == target) ? 2 : 0;
            vals.push_back(presum - i);
        }
        sort(vals.begin(), vals.end());
        vals.erase(unique(vals.begin(), vals.end()), vals.end());
        
        unordered_map<int, int> val2pos;
        for (int i = 0; i < vals.size(); i++) {
            val2pos[vals[i]] = i + 1; // 映射到 1~m
        }
        
        // 2. 树状数组统计
        Fenwick tr(vals.size() + 5);
        tr.add(val2pos[0], 1); // 初始化空数组的状态
        
        LL ans = 0;
        int cur = 0;
        for (int i = 1; i <= n; i++) {
            cur += (nums[i - 1] == target) ? 2 : 0;
            int index = val2pos[cur - i];
            
            // 查询有多少个之前的前缀和严格小于当前值
            ans += tr.query(index - 1); 
            
            // 将当前状态加入树状数组
            tr.add(index, 1);
        }
        return ans;
    }
};
